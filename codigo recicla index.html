                <p id="previewText" class="text-gray-400">La imagen subida aparecerá aquí.</p>

                <img id="uploadedImage" class="mt-4 max-h-96 w-auto mx-auto rounded-lg shadow-lg hidden" alt="Imagen cargada para escanear">

            </div>

                    <span id="buttonText">🔍 Escanear Material</span>

                </button>

            <div id="errorContainer" class="hidden p-4 bg-red-100 border border-red-400 text-red-700 rounded-lg" role="alert">

    <!-- Módulo de JavaScript para la lógica de la aplicación -->

    <script type="module">

        // --- Constantes y Variables Globales del Entorno (Requeridas por Canvas) ---

        const apiKey = ""; 

        const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-05-20:generateContent?key=${apiKey}`;

        const MAX_IMAGE_WIDTH = 1024; // Máximo de píxeles para el lado más largo de la imagen.

 

        // --- Elementos del DOM ---

        const imageUpload = document.getElementById('imageUpload');

        const uploadedImage = document.getElementById('uploadedImage');

        const previewText = document.getElementById('previewText');

        const scanButton = document.getElementById('scanButton');

        const loadingSpinner = document.getElementById('loadingSpinner');

        const buttonText = document.getElementById('buttonText');

        const resultContent = document.getElementById('resultContent');

        const errorContainer = document.getElementById('errorContainer');

        const errorMessage = document.getElementById('errorMessage');

 

        let base64Image = null;

        let imageMimeType = null;

 

        // --- Funciones de Utilidad ---

 

        /**

         * Muestra/Oculta el estado de carga (spinner y botón).

         * @param {boolean} isLoading - Si es verdadero, muestra el spinner.

         */

        function setLoading(isLoading) {

            scanButton.disabled = isLoading || !base64Image;

            if (isLoading) {

                loadingSpinner.style.display = 'block';

                buttonText.textContent = 'Analizando...';

            } else {

                loadingSpinner.style.display = 'none';

                buttonText.textContent = '🔍 Escanear Material';

            }

        }

 

        /**

         * Muestra un mensaje de error.

         * @param {string} message - El mensaje de error a mostrar.

         */

        function showError(message) {

            errorMessage.textContent = message;

            errorContainer.classList.remove('hidden');

        }

 

        /**

         * Convierte el archivo de imagen a una cadena Base64 REDIMENSIONADA.

         * Esto previene que imágenes grandes saturen la API o fallen por tiempo de espera.

         * @param {File} file - El archivo de imagen.

         * @returns {Promise<string>} Promesa que resuelve la cadena Base64 (sin prefijo).

         */

        function resizeImage(file) {

            return new Promise((resolve, reject) => {

                const reader = new FileReader();

                reader.onload = (event) => {

                    const img = new Image();

                    img.onload = () => {

                        let width = img.width;

                        let height = img.height;

                        

                        // Determinar si se necesita redimensionar

                        if (width > MAX_IMAGE_WIDTH || height > MAX_IMAGE_WIDTH) {

                            if (width > height) {

                                height *= MAX_IMAGE_WIDTH / width;

                                width = MAX_IMAGE_WIDTH;

                            } else {

                                width *= MAX_IMAGE_WIDTH / height;

                                height = MAX_IMAGE_WIDTH;

                            }

                        }

                        

                        // Crear el canvas y dibujar la imagen redimensionada

                        const canvas = document.createElement('canvas');

                        canvas.width = width;

                        canvas.height = height;

                        const ctx = canvas.getContext('2d');

                        ctx.drawImage(img, 0, 0, width, height);

 

                        // Convertir el canvas a Base64 (calidad 0.8 para compresión)

                        const resizedDataUrl = canvas.toDataURL(file.type, 0.8);

                        resolve(resizedDataUrl.split(',')[1]); // Solo la parte base64

                    };

                    img.onerror = error => reject("Error al cargar la imagen para redimensionar.");

                    img.src = event.target.result;

                };

                reader.onerror = error => reject("Error al leer el archivo.");

                reader.readAsDataURL(file);

            });

        }

 

        // --- Lógica del Evento de Carga de Imagen ---

 

        imageUpload.addEventListener('change', async (event) => {

            errorContainer.classList.add('hidden');

            const file = event.target.files[0];

            if (file) {

                try {

                    // Muestra un mensaje de procesamiento

                    resultContent.innerHTML = `

                        <div class="flex items-center justify-center text-blue-600">

                            <div class="spinner mr-3 !static !inline-block !border-t-blue-500"></div>

                            <span>Procesando imagen (reduciendo tamaño)...</span>

                        </div>

                    `;

                    scanButton.disabled = true; // Deshabilita el botón mientras se procesa

 

                    // 1. Obtener MIME Type

                    imageMimeType = file.type;

 

                    // 2. Redimensionar y obtener Base64

                    base64Image = await resizeImage(file);

 

                    // 3. Muestra la imagen (desde el objeto blob original para previsualización)

                    const imageUrl = URL.createObjectURL(file);

                    uploadedImage.src = imageUrl;

                    uploadedImage.classList.remove('hidden');

                    previewText.classList.add('hidden');

                    

                    scanButton.disabled = false;

                    resultContent.innerHTML = '<p>✅ Imagen procesada. Presiona "Escanear Material".</p>';

 

                } catch (error) {

                    showError('Error al procesar la imagen: ' + (error.message || error));

                    console.error('Error al cargar/redimensionar la imagen:', error);

                    base64Image = null;

                    scanButton.disabled = true;

                    uploadedImage.classList.add('hidden');

                    previewText.classList.remove('hidden');

                    resultContent.innerHTML = '<p class="text-red-500">Hubo un problema al cargar la imagen.</p>';

                }

            } else {

                uploadedImage.classList.add('hidden');

                previewText.classList.remove('hidden');

                base64Image = null;

                scanButton.disabled = true;

            }

        });

 

        // --- Lógica del Llamado a la API de Gemini ---

 

        scanButton.addEventListener('click', async () => {

            if (!base64Image || !imageMimeType) {

                showError('Por favor, selecciona una imagen primero.');

                return;

            }

 

            setLoading(true);

            errorContainer.classList.add('hidden');

            resultContent.innerHTML = `

                <div class="flex items-center justify-center text-emerald-600">

                    <div class="spinner mr-3 !static !inline-block"></div>

                    <span>La IA está analizando el material...</span>

                </div>

            `;

 

            // Configuración del request a la API

            const systemPrompt = "Eres un Asistente de Reciclaje. Tu tarea es analizar la imagen de un producto o residuo e identificar su material principal. Debes clasificarlo estrictamente en una de las siguientes categorías: **Lata, Cartón, Vidrio, Plástico, o Otro/Mixto**. Después de la clasificación, proporciona una recomendación concisa sobre cómo debe reciclarse ese objeto específico en un contexto general (ej. 'Enjuagar y depositar en el contenedor amarillo'). Responde solo con esta información.";

            const userQuery = "Clasifica el material principal de este objeto y dime la mejor acción de reciclaje.";

 

            const payload = {

                contents: [{

                    role: "user",

                    parts: [

                        { text: userQuery },

                        {

                            inlineData: {

                                mimeType: imageMimeType, // Usamos el MIME type original

                                data: base64Image

                            }

                        }

                    ]

                }],

                systemInstruction: {

                    parts: [{ text: systemPrompt }]

                }

            };

 

            const maxRetries = 5;

            for (let attempt = 0; attempt < maxRetries; attempt++) {

                try {

                    const response = await fetch(apiUrl, {

                        method: 'POST',

                        headers: { 'Content-Type': 'application/json' },

                        body: JSON.stringify(payload)

                    });

 

                    if (!response.ok) {

                        if (response.status === 429 && attempt < maxRetries - 1) {

                            // Implementar backoff exponencial (1s, 2s, 4s, ...)

                            const delay = Math.pow(2, attempt) * 1000;

                            await new Promise(resolve => setTimeout(resolve, delay));

                            continue; // Intentar de nuevo

                        }

                        const errorData = await response.json();

                        throw new Error(`Error en la API: ${response.status} - ${errorData.error?.message || 'Error desconocido'}`);

                    }

 

                    const result = await response.json();

                    const aiText = result.candidates?.[0]?.content?.parts?.[0]?.text;

 

                    if (aiText) {

                        // Formateo de la respuesta para el usuario

                        const formattedText = aiText.replace(/(\*\*[^*]+\*\*)/g, '<span class="text-emerald-700 font-bold">$1</span>')

                                                    .replace(/\n/g, '<br>');

 

                        resultContent.innerHTML = `

                            <p class="text-xl leading-relaxed">${formattedText}</p>

                        `;

                    } else {

                        throw new Error('La respuesta de la IA estaba vacía o mal formada.');

                    }

                    

                    break; // Salir del bucle si es exitoso

 

                } catch (error) {

                    if (attempt === maxRetries - 1) {

                        showError('No se pudo completar la solicitud después de varios intentos. Puede ser un problema de red o que la imagen aún es muy compleja.');

                        resultContent.innerHTML = '<p class="text-red-500">Ocurrió un error. Verifica tu conexión o intenta con otra imagen.</p>';

                        console.error('Error durante el análisis:', error);

                    }

                }

            }

 

            setLoading(false);

        });

 

        async function setupFirebase() {

            try {

                // No se requiere inicialización de Firebase para esta tarea específica.

            } catch (e) {

                console.warn("Firebase no se inicializó, pero la función de escaneo de IA funcionará independientemente.");

            }

        }

        setupFirebase();

 

    </script>
